const AUTHORISED_SHEET = "Authorised"; // tab name const STATUS_COL = 6; // column F (update if your Status column differs) function doPost(e) { try { const body = JSON.parse(e.postData.contents); if (body.action === "verify") return handleVerify(body); if (body.action === "update") return handleUpdate(body); return jsonResp({ ok: false, error: "Unknown action" }); } catch(err) { return jsonResp({ ok: false, error: err.message }); } } function handleVerify(body) { const email = getVerifiedEmail(body); if (!email) return jsonResp({ ok: false, error: "Invalid or expired token" }); const authorised = isAuthorised(email); return jsonResp({ ok: true, authorised, email }); } function handleUpdate(body) { // Re-verify on every write — never trust the browser's cached claim const email = getVerifiedEmail(body); if (!email) return jsonResp({ ok: false, error: "Invalid or expired token" }); if (!isAuthorised(email)) return jsonResp({ ok: false, error: "Not authorised" }); const row = parseInt(body.rowIndex, 10); const status = body.newStatus; // Exact values stored in the sheet — underscores, no spaces const allowed = ["Complete", "In_Progress", "Planned", "Not_Started"]; if (!allowed.includes(status)) return jsonResp({ ok: false, error: "Invalid status value: " + status }); if (isNaN(row) || row < 2) return jsonResp({ ok: false, error: "Invalid row index" }); SpreadsheetApp.getActiveSpreadsheet() .getSheets()[0] .getRange(row, STATUS_COL) .setValue(status); return jsonResp({ ok: true }); } function getVerifiedEmail(body) { try { if (body.idToken) { const res = UrlFetchApp.fetch( "https://oauth2.googleapis.com/tokeninfo?id_token=" + body.idToken, { muteHttpExceptions: true } ); if (res.getResponseCode() !== 200) return null; const info = JSON.parse(res.getContentText()); if (info.error || !info.email_verified) return null; return info.email.toLowerCase(); } if (body.accessToken) { const res = UrlFetchApp.fetch( "https://www.googleapis.com/oauth2/v3/userinfo", { headers: { Authorization: "Bearer " + body.accessToken }, muteHttpExceptions: true } ); if (res.getResponseCode() !== 200) return null; const info = JSON.parse(res.getContentText()); return info.email ? info.email.toLowerCase() : null; } return null; } catch(e) { return null; } } function isAuthorised(email) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(AUTHORISED_SHEET); if (!sheet) return false; return sheet.getDataRange().getValues() .flat() .some(v => String(v).trim().toLowerCase() === email); } function jsonResp(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
